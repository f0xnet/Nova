Conversation

Système de dialogues — afficher une suite de répliques, faire des choix, brancher l'histoire selon les flags et l'inventaire.

Global automatique

API

FonctionDescription
Conversation.start(dialogue)Lance une conversation à partir d'une table de répliques.
Conversation.stop()Ferme la boîte de dialogue immédiatement.
Conversation.isActive()Vrai si un dialogue est en cours.
Conversation.onEnd(fonction)Appelée quand le dialogue se termine (dernier choix ou dernière réplique).

Structure d'un dialogue

Un dialogue est un tableau de nœuds. Chaque nœud représente une réplique ou un choix.

ChampDescription
speakerNom du personnage qui parle.
textTexte de la réplique.
choicesListe de choix (optionnel). Si absent : bouton "Suivant" automatique.
nextIndex du prochain nœud. Si absent : continue au suivant.
conditionFonction qui renvoie vrai/faux pour afficher un choix.
actionFonction appelée quand ce nœud est atteint.

Exemples

lua  ·  Dialogue simple avec un PNJ
local dialogue_marchand = {
    { speaker = "Marchand", text = "Bonjour voyageur ! Que cherches-tu ?" },
    { speaker = "Marchand", text = "J'ai de belles potions si tu es intéressé." },
    { speaker = "Marchand", text = "Reviens quand tu veux !" },
}

-- Lancer le dialogue quand le joueur interagit
function OnTriggerEnter(entity, other)
    if other:hasTag("player") then
        Conversation.start(dialogue_marchand)
    end
end

lua  ·  Dialogue avec choix et branches
local dialogue_garde = {
    -- Nœud 1 : première réplique
    {
        speaker = "Garde",
        text    = "Halte ! Qui vas-tu ? Que veux-tu ?",
        choices = {
            { text = "Je cherche le roi.",      next = 2 },
            { text = "Juste passer mon chemin.", next = 3 },
            { text = "Je t'attaque !",           next = 4 },
        }
    },

    -- Nœud 2 : joueur veut voir le roi
    {
        speaker = "Garde",
        text    = "Bien. Suivez le couloir à gauche.",
        action  = function()
            Flag.set("garde_parle")
        end
    },

    -- Nœud 3 : joueur passe
    {
        speaker = "Garde",
        text    = "Très bien. Mais ne faites pas d'histoires.",
    },

    -- Nœud 4 : joueur attaque
    {
        speaker = "Garde",
        text    = "Quoi ?! Gardes, à moi !",
        action  = function()
            EventBus.emit("alerte_gardes", {})
        end
    },
}

-- Lancer le dialogue
Conversation.start(dialogue_garde)

lua  ·  Choix conditionnels (selon inventaire ou flags)
local dialogue_sage = {
    {
        speaker = "Sage",
        text    = "Tu es revenu... que veux-tu ?",
        choices = {
            -- Ce choix n'apparaît que si le joueur a la gemme
            {
                text      = "Je t'apporte la Gemme de Feu.",
                next      = 2,
                condition = function()
                    return Inventory.has("gemme_feu")
                end
            },
            -- Ce choix n'apparaît que si le joueur a déjà parlé au roi
            {
                text      = "Le roi m'a parlé de toi.",
                next      = 3,
                condition = function()
                    return Flag.get("roi_parle")
                end
            },
            -- Ce choix est toujours visible
            { text = "Rien, au revoir.", next = 4 },
        }
    },

    { speaker = "Sage", text = "Excellent ! Tu as réussi la quête !", action = function()
        Quest.complete("trouver_gemme")
    end },

    { speaker = "Sage", text = "Ah oui, Sa Majesté m'a prévenu de ta venue." },

    { speaker = "Sage", text = "Comme tu veux, voyageur." },
}

Conversation.start(dialogue_sage)

-- Action après la fin du dialogue
Conversation.onEnd(function()
    print("Dialogue avec le sage terminé.")
end)
