Class

Créer des classes Lua réutilisables — organiser son code avec l'orienté objet simple : héritage, constructeur, méthodes.

Global automatique

API

FonctionDescription
Class.new(nom, parent)Crée une nouvelle classe. parent est optionnel (héritage).
MaClasse:new(...)Crée une instance. Appelle init(...) si défini.
MaClasse:extend(nom)Crée une sous-classe qui hérite de MaClasse.
instance:is(Classe)Vrai si l'instance est de ce type (ou d'un parent).

Exemples

lua  ·  Classe simple
-- Définir une classe "Arme"
local Arme = Class.new("Arme")

function Arme:init(nom, degats)
    self.nom    = nom
    self.degats = degats
end

function Arme:attaquer(cible)
    Stats.mod(cible.id, "Vie", -self.degats)
    print(self.nom .. " inflige " .. self.degats .. " dégâts !")
end

function Arme:getDescription()
    return self.nom .. " (" .. self.degats .. " dégâts)"
end

-- Utiliser la classe
local epee   = Arme:new("Épée longue",  25)
local hache  = Arme:new("Hache de fer", 35)

epee:attaquer(ennemi)
print(hache:getDescription())

lua  ·  Héritage
-- Sous-classe "ArmeEnchantee" qui hérite de "Arme"
local ArmeEnchantee = Arme:extend("ArmeEnchantee")

function ArmeEnchantee:init(nom, degats, element)
    Arme.init(self, nom, degats)  -- appeler le parent
    self.element = element
end

function ArmeEnchantee:attaquer(cible)
    Arme.attaquer(self, cible)    -- dégâts normaux

    -- Effet supplémentaire selon l'élément
    if self.element == "feu" then
        Effect.apply(cible, "brulure", 5.0)
    elseif self.element == "glace" then
        Effect.apply(cible, "ralenti", 3.0)
    end
end

-- Utiliser
local baton_feu  = ArmeEnchantee:new("Bâton de feu",  30, "feu")
local epee_glace = ArmeEnchantee:new("Épée de glace", 20, "glace")

baton_feu:attaquer(ennemi)

-- Vérifier le type
print(baton_feu:is(Arme))          -- true (héritage)
print(baton_feu:is(ArmeEnchantee)) -- true
