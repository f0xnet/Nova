Scene

Gestion des scènes — charger, changer, recharger. Utilisez toujours SceneFX.transition pour un fondu lors des changements.

Global automatique

API

FonctionDescription
Scene.setActive(nom)Change de scène immédiatement (sans fondu).
Scene.transition(nom, opts)Change de scène avec fondu. opts.fade et opts.duration.
Scene.reload()Recharge la scène actuelle (utile après une mort).
Scene.current()Nom de la scène actuellement active.
Scene.has(nom)Vrai si la scène est chargée.

Exemples

lua  ·  Changer de scène avec fondu
-- Méthode recommandée : fondu enchaîné
function entrerDansLeDonjon()
    Scene.transition("donjon_niveau1", {
        fade     = true,
        duration = 0.6,
    })
end

-- Ou via SceneFX pour plus de contrôle
function allerVersLaVille()
    SceneFX.transition(function()
        -- Code exécuté quand l'écran est noir
        Scene.setActive("ville_principale")
    end, 0.5)
end

-- Déclencher depuis un trigger
function OnTriggerEnter(triggerId, entityId)
    local joueur = World.findByTag("player")
    if joueur and entityId == joueur.id then
        entrerDansLeDonjon()
    end
end

lua  ·  Recharger après une mort
function joueurMort()
    SceneFX.fadeIn(1.5, function()
        -- Remettre à zéro les stats
        Stats.set(joueur.id, "Vie", 100)

        -- Revenir au dernier checkpoint
        local cp = World.findByTag("checkpoint_actif")
        if cp then
            World.setPosition(joueur, cp:getPosition().x, cp:getPosition().y)
        else
            -- Pas de checkpoint : recharger la scène entière
            Scene.reload()
        end

        SceneFX.fadeOut(1.0)
    end)
end

lua  ·  Réagir au chargement d'une scène
-- Réagir au changement de scène via EventBus
EventBus.on("scene_changed", function(data)
    local nom = data.name  -- nom de la nouvelle scène
    print("Scène chargée : " .. nom)

    -- Jouer une musique différente selon la scène
    if nom == "donjon_niveau1" then
        Sound.playMusic("data/music/donjon.ogg")
    elseif nom == "ville_principale" then
        Sound.playMusic("data/music/ville.ogg")
    end
end)

-- Connaître la scène actuelle
EventBus.on("scene_changed", function(data)
    print("Changement vers : " .. Scene.current())
end)
