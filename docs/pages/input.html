Input

Lire les entrées clavier, souris et manette — touches pressées, axes de direction, position de la souris.

Global automatique

API — Clavier

FonctionDescription
Input.isDown(touche)Vrai tant que la touche est maintenue.
Input.isPressed(touche)Vrai uniquement le premier frame où la touche est pressée.
Input.isReleased(touche)Vrai uniquement le frame où la touche est relâchée.

API — Souris

FonctionDescription
Input.mouseX()Position X de la souris dans le monde.
Input.mouseY()Position Y de la souris dans le monde.
Input.mouseScreenX()Position X à l'écran (pixels).
Input.mouseScreenY()Position Y à l'écran (pixels).
Input.isDown("mouse_left")Clic gauche maintenu.
Input.isPressed("mouse_right")Clic droit — premier frame seulement.

API — Manette (InputEx)

FonctionDescription
InputEx.axisX()Axe horizontal du stick gauche (−1.0 à +1.0).
InputEx.axisY()Axe vertical du stick gauche (−1.0 à +1.0).
InputEx.isDown(bouton)Bouton manette maintenu : "A", "B", "X", "Y", "LB", "RB"…
InputEx.isPressed(bouton)Bouton — premier frame seulement.
InputEx.vibrate(gauche, droite, duree)Vibration des moteurs (0.0 à 1.0).
InputEx.isConnected()Vrai si une manette est branchée.

Touches communes

"space", "enter", "escape", "tab", "shift", "ctrl", "alt"
"up", "down", "left", "right"
"a" … "z", "0" … "9"
"f1" … "f12", "mouse_left", "mouse_right", "mouse_middle"

Exemples

lua  ·  Déplacement clavier 4 directions
local VITESSE = 200

function OnUpdate(entity, dt)
    local dx, dy = 0, 0

    if Input.isDown("left")  or Input.isDown("a") then dx = -1 end
    if Input.isDown("right") or Input.isDown("d") then dx =  1 end
    if Input.isDown("up")    or Input.isDown("w") then dy = -1 end
    if Input.isDown("down")  or Input.isDown("s") then dy =  1 end

    -- Normaliser pour ne pas aller plus vite en diagonale
    if dx ~= 0 and dy ~= 0 then
        dx = dx * 0.707
        dy = dy * 0.707
    end

    local pos = World.getPosition(entity)
    World.setPosition(entity, pos.x + dx * VITESSE * dt,
                               pos.y + dy * VITESSE * dt)
end

lua  ·  Attaque au clic souris
function OnUpdate(entity, dt)
    -- Tirer vers la souris au clic gauche
    if Input.isPressed("mouse_left") then
        local pos    = World.getPosition(entity)
        local cible_x = Input.mouseX()
        local cible_y = Input.mouseY()

        Projectile.fire(pos.x, pos.y, {
            dx       = cible_x - pos.x,
            dy       = cible_y - pos.y,
            vitesse  = 400,
            prefab   = "projectile_boule_de_feu",
        })
    end
end

lua  ·  Déplacement manette avec vibration
local VITESSE = 200

function OnUpdate(entity, dt)
    local dx = InputEx.axisX()
    local dy = InputEx.axisY()

    local pos = World.getPosition(entity)
    World.setPosition(entity, pos.x + dx * VITESSE * dt,
                               pos.y + dy * VITESSE * dt)

    -- Saut au bouton A
    if InputEx.isPressed("A") then
        EventBus.emit("joueur_saute", {})
    end
end

-- Vibrer quand le joueur est touché
EventBus.on("joueur_touche", function()
    InputEx.vibrate(0.8, 0.4, 0.3)
end)
