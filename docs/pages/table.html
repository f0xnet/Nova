Table

Fonctions utilitaires pour les tableaux Lua — chercher, filtrer, transformer, trier des listes.

Global automatique

API

FonctionDescription
Table.contains(t, valeur)Vrai si la valeur est dans le tableau.
Table.find(t, fonction)Retourne le premier élément pour lequel fonction(elem) est vrai.
Table.filter(t, fonction)Retourne un nouveau tableau avec seulement les éléments valides.
Table.map(t, fonction)Retourne un nouveau tableau transformé par la fonction.
Table.count(t, fonction)Nombre d'éléments pour lesquels la fonction est vraie.
Table.sum(t, clé)Somme de elem[clé] pour tous les éléments.
Table.sort(t, fonction)Trie le tableau sur place selon une fonction de comparaison.
Table.copy(t)Copie superficielle du tableau.
Table.remove(t, valeur)Retire la première occurrence de la valeur.
Table.clear(t)Vide le tableau.

Exemples

lua  ·  Filtrer et chercher
local ennemis = World.findAllByTag("enemy")

-- Garder seulement les ennemis encore en vie
local ennemis_vivants = Table.filter(ennemis, function(e)
    return Stats.get(e.id, "Vie", 0) > 0
end)

-- Trouver l'ennemi le plus proche du joueur
local pos_joueur = World.getPosition(joueur)
local plus_proche = Table.find(ennemis, function(e)
    local pos = World.getPosition(e)
    return Vec2.distance(Vec2.new(pos_joueur.x, pos_joueur.y), Vec2.new(pos.x, pos.y)) < 200
end)

-- Vérifier si une liste contient un item
local inventaire = Inventory.getAll(joueur)
if Table.contains(inventaire, "cle_donjon") then
    print("Le joueur a la clé !")
end

lua  ·  Trier et transformer
local ennemis = World.findAllByTag("enemy")

-- Trier par distance au joueur (du plus proche au plus loin)
local pos_j = World.getPosition(joueur)
Table.sort(ennemis, function(a, b)
    local pa = World.getPosition(a)
    local pb = World.getPosition(b)
    return Vec2.distance(Vec2.new(pos_j.x, pos_j.y), Vec2.new(pa.x, pa.y))
         < Vec2.distance(Vec2.new(pos_j.x, pos_j.y), Vec2.new(pb.x, pb.y))
end)

-- Créer une liste de positions
local positions = Table.map(ennemis, function(e)
    return World.getPosition(e)
end)

-- Compter les ennemis avec moins de 20 PV
local nb_blesses = Table.count(ennemis, function(e)
    return Stats.get(e.id, "Vie", 0) < 20
end)
