InputBind

Raccourcis configurables — associer des actions nommées à des touches et permettre le remapping.

Global automatique

API

FonctionDescription
InputBind.register(action, touche)Enregistre une action avec sa touche par défaut.
InputBind.remap(action, touche)Change la touche de l'action.
InputBind.isPressed(action)Vrai tant que la touche est maintenue.
InputBind.isJustPressed(action)Vrai uniquement le premier frame de la pression.
InputBind.isJustReleased(action)Vrai uniquement le frame du relâchement.
InputBind.getKey(action)Retourne la touche actuellement assignée.
InputBind.getAll()Tableau de toutes les actions et leurs touches.
InputBind.save()Retourne une table des bindings (pour la sauvegarder).
InputBind.load(table)Restaure des bindings depuis une table sauvegardée.

Exemples

lua  ·  Définir et utiliser des actions
-- Définir les actions une seule fois (dans un script de démarrage)
InputBind.register("bouger_gauche",  "Left")
InputBind.register("bouger_droite",  "Right")
InputBind.register("bouger_haut",    "Up")
InputBind.register("bouger_bas",     "Down")
InputBind.register("attaquer",       "Space")
InputBind.register("interagir",      "E")
InputBind.register("inventaire",     "I")

-- Utiliser les actions dans le script joueur
function update(entity, dt)
    local dx, dy = 0, 0

    if InputBind.isPressed("bouger_gauche") then dx = -1 end
    if InputBind.isPressed("bouger_droite") then dx =  1 end
    if InputBind.isPressed("bouger_haut")   then dy = -1 end
    if InputBind.isPressed("bouger_bas")    then dy =  1 end

    if InputBind.isJustPressed("attaquer")  then attaquer(entity) end
end

lua  ·  Sauvegarder et restaurer les bindings
-- Remapper une touche depuis le menu d'options
function changerTouche(action, nouvelle_touche)
    InputBind.remap(action, nouvelle_touche)

    -- Sauvegarder dans Persist
    local bindings = InputBind.save()
    Persist.set("bindings", Data.encode(bindings))
    Notify.show("Touche modifiée : " .. nouvelle_touche)
end

-- Restaurer les bindings au démarrage
local bindings_json = Persist.get("bindings")
if bindings_json then
    InputBind.load(Data.decode(bindings_json))
end
