Physics

Appliquer des forces et des impulsions aux entités — lancer, repousser, simuler de la gravité ou des rebonds.

Global automatique

API

FonctionDescription
Physics.setVelocity(entity, vx, vy)Définit la vitesse instantanée en pixels/seconde.
Physics.getVelocity(entity)Retourne {x, y} — la vitesse actuelle.
Physics.addForce(entity, fx, fy)Applique une force continue (s'additionne chaque frame).
Physics.addImpulse(entity, ix, iy)Applique une impulsion instantanée (poussée ponctuelle).
Physics.setGravity(entity, gy)Gravité personnalisée pour cette entité (pixels/s²).
Physics.setDamping(entity, d)Frottement (0.0 = sans frottement, 1.0 = stop immédiat).
Physics.isGrounded(entity)Vrai si l'entité touche le sol.

Exemples

lua  ·  Saut et gravité
local FORCE_SAUT = -500  -- vers le haut (y négatif)

function OnInit(entity)
    Physics.setGravity(entity, 800)   -- gravité forte
    Physics.setDamping(entity, 0.1)   -- léger frottement au sol
end

function OnUpdate(entity, dt)
    -- Sauter quand on appuie sur espace et qu'on est au sol
    if InputBind.isPressed("sauter") and Physics.isGrounded(entity) then
        Physics.addImpulse(entity, 0, FORCE_SAUT)
        Anim.play(entity, "jump", false)
        Sound.play("data/sounds/saut.ogg")
    end
end

lua  ·  Knockback (repousser un ennemi)
-- Repousser un ennemi dans la direction opposée au joueur
function knockback(ennemi, joueur, force)
    local pos_j = World.getPosition(joueur)
    local pos_e = World.getPosition(ennemi)

    -- Direction de la poussée (de joueur vers ennemi)
    local dir = Vec2.normalize(Vec2.sub(
        Vec2.new(pos_e.x, pos_e.y),
        Vec2.new(pos_j.x, pos_j.y)
    ))

    Physics.addImpulse(ennemi, dir.x * force, dir.y * force)
end

-- Quand le joueur attaque
knockback(ennemi, joueur, 400)

lua  ·  Projectile avec gravité
-- Flèche qui suit une parabole
function tirerFleche(x, y, angle_deg)
    local proj = Pool.acquire("fleche")
    World.setPosition(proj, x, y)

    local dir     = Vec2.fromAngle(angle_deg)
    local vitesse = 300
    Physics.setVelocity(proj, dir.x * vitesse, dir.y * vitesse)
    Physics.setGravity(proj, 400)   -- retombe naturellement
end
