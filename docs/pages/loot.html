Loot

Butin aléatoire — définir des tables de loot avec probabilités et faire tomber des objets à la mort d'un ennemi.

Global automatique

API

FonctionDescription
Loot.roll(table)Tire au sort un objet dans la table de loot. Retourne l'id de l'objet ou nil.
Loot.rollAll(table)Teste chaque entrée séparément (plusieurs objets possibles). Retourne un tableau d'ids.
Loot.drop(x, y, table)Fait apparaître les objets dans le monde à la position (x, y).

Structure d'une table de loot

Chaque entrée a un id (item), un poids (chance relative) et une quantite optionnelle.

lua  ·  Définir une table de loot
-- Table de loot d'un gobelin
local loot_gobelin = {
    { id = "piece_or",     poids = 60, quantite = { min = 1, max = 5 } },
    { id = "potion_soin",  poids = 25 },
    { id = "cle_rouille",  poids = 10 },
    { id = "epee_gobelin", poids = 5  },
    -- poids total = 100 → probabilités directement en %
}

-- Table avec chances indépendantes (rollAll)
local loot_coffre = {
    { id = "piece_or",    chance = 1.0,  quantite = { min = 10, max = 30 } },
    { id = "potion_soin", chance = 0.5  },  -- 50% de chance
    { id = "epee_rare",   chance = 0.05 },  -- 5% de chance
}

Exemples

lua  ·  Loot à la mort d'un ennemi
local loot_gobelin = {
    { id = "piece_or",     poids = 60, quantite = { min = 1, max = 5 } },
    { id = "potion_soin",  poids = 25 },
    { id = "cle_rouille",  poids = 10 },
    { id = "epee_gobelin", poids = 5  },
}

function OnDestroy(entity)
    local pos = World.getPosition(entity)

    -- Fait tomber 1 objet selon les poids
    Loot.drop(pos.x, pos.y, loot_gobelin)

    -- Ou récupérer l'objet tiré sans le faire apparaître
    local objet = Loot.roll(loot_gobelin)
    if objet then
        print("Le gobelin a lâché : " .. objet)
    end
end

lua  ·  Coffre avec plusieurs objets possibles
local loot_coffre_boss = {
    { id = "piece_or",    chance = 1.0,  quantite = { min = 50, max = 100 } },
    { id = "potion_soin", chance = 0.75 },
    { id = "armure_rare", chance = 0.15 },
    { id = "cle_donjon",  chance = 0.5  },
}

function OnInteract(entity, joueur)
    local pos = World.getPosition(entity)

    -- rollAll : chaque item est testé séparément
    local objets = Loot.rollAll(loot_coffre_boss)

    for _, id in ipairs(objets) do
        Inventory.give(joueur, id)
        Notify.show("Obtenu : " .. id, "success")
    end

    entity:disable()  -- fermer le coffre
end
