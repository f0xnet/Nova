Inventory

Gérer les objets portés par chaque entité — ramasser, utiliser, transférer. Inspiré du système d'inventaire Papyrus / Skyrim.

Global automatiqueÉmet des événements

API

FonctionDescriptionRetour
Inventory.add(entityId, item, qte)Ajoute des objets. qte = 1 par défaut.qté ajoutée
Inventory.remove(entityId, item, qte)Retire des objets. Ne plante pas si insuffisant.qté retirée
Inventory.count(entityId, item)Combien de cet objet l'entité possède-t-elle ?number
Inventory.has(entityId, item, qte)Possède au moins qte de cet objet ? (qte=1 par défaut)bool
Inventory.getAll(entityId)Tous les objets : { item = qte, ... }table
Inventory.transfer(from, to, item, qte)Transfère des objets d'une entité à une autre.—
Inventory.defineItem(item, def)Définit les propriétés d'un type d'objet (poids, stack max, nom…).—
Inventory.clear(entityId)Vide complètement l'inventaire.—

Exemple — Coffre et clé

lua  ·  scripts/coffre.lua
local estOuvert = false

function OnInit(entity, props)
    -- Définir le contenu du coffre
    Inventory.add(entity.id, "piece_or",  props.or  or 25)
    Inventory.add(entity.id, "potion_vie", props.potions or 1)
end

function OnMessage(entity, msg, data)
    if msg == "interagir" then
        local joueur = World.findByTag("player")
        if not joueur then return end

        if estOuvert then
            Notify.show("Ce coffre est déjà ouvert.", { duration = 2 })
            return
        end

        -- Vérifier si le joueur a la clé
        if Inventory.has(joueur.id, "cle_or") then
            Inventory.remove(joueur.id, "cle_or", 1)

            -- Transférer tout le contenu du coffre au joueur
            local contenu = Inventory.getAll(entity.id)
            for item, qte in pairs(contenu) do
                Inventory.transfer(entity.id, joueur.id, item, qte)
                Notify.show("+" .. qte .. " " .. item, { color = "gold" })
            end

            Anim.play(entity, "open", false)
            Sound.play("data/sounds/coffre.ogg")
            estOuvert = true
        else
            Notify.show("Il vous faut une clé dorée.", { color = "red", duration = 2 })
            Sound.play("data/sounds/locked.ogg")
        end
    end
end

Exemple — Utiliser une potion

lua
function utiliserPotion(joueur)
    if not Inventory.has(joueur.id, "potion_vie") then
        Notify.show("Pas de potion de vie !", { color = "red" })
        return
    end

    Inventory.remove(joueur.id, "potion_vie", 1)

    local soin = 50
    Stats.mod(joueur.id, "Vie", soin)
    Notify.floatingText(joueur, "+" .. soin .. " PV", { color = "lime" })
    Sound.play("data/sounds/potion.ogg")
    Particles.emit(World.getPosition(joueur).x, World.getPosition(joueur).y, {
        count = 15, speed = 60, color = Color.new(0, 220, 80), lifetime = 0.8
    })
end
