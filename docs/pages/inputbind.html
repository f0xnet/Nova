InputBind

Raccourcis configurables — définir des actions nommées, les associer à des touches, permettre à l'utilisateur de les remapper.

Global automatique

API

FonctionDescription
InputBind.define(action, touches)Définit une action et ses touches par défaut. touches = liste de strings.
InputBind.isDown(action)Vrai si l'une des touches de l'action est maintenue.
InputBind.isPressed(action)Vrai uniquement le premier frame où l'action est déclenchée.
InputBind.isReleased(action)Vrai quand l'action est relâchée.
InputBind.remap(action, touche)Change la touche principale de l'action.
InputBind.reset(action)Remet la touche par défaut.
InputBind.getKey(action)Retourne la touche principale actuelle.

Exemples

lua  ·  Définir et utiliser des actions
-- Définir les actions une seule fois (dans un script de démarrage)
InputBind.define("bouger_gauche",  { "left",  "a"       })
InputBind.define("bouger_droite",  { "right", "d"       })
InputBind.define("bouger_haut",    { "up",    "w"       })
InputBind.define("bouger_bas",     { "down",  "s"       })
InputBind.define("attaquer",       { "space", "mouse_left" })
InputBind.define("dash",           { "shift" })
InputBind.define("interagir",      { "e",    "enter"    })
InputBind.define("inventaire",     { "i",    "tab"      })

-- Utiliser les actions dans le script joueur
function OnUpdate(entity, dt)
    local dx, dy = 0, 0

    if InputBind.isDown("bouger_gauche") then dx = -1 end
    if InputBind.isDown("bouger_droite") then dx =  1 end
    if InputBind.isDown("bouger_haut")   then dy = -1 end
    if InputBind.isDown("bouger_bas")    then dy =  1 end

    if InputBind.isPressed("attaquer")   then attaquer(entity) end
    if InputBind.isPressed("dash")       then dash(entity)     end
end

lua  ·  Permettre de remapper les touches
-- Afficher la touche actuelle dans le menu
local touche_attaque = InputBind.getKey("attaquer")
print("Attaque : " .. touche_attaque)  -- ex: "space"

-- Remapper depuis le menu d'options
function OnToucheChoisie(action, nouvelle_touche)
    InputBind.remap(action, nouvelle_touche)

    -- Sauvegarder dans Persist pour la prochaine session
    Persist.set("bind_" .. action, nouvelle_touche)
    Notify.show("Touche modifiée : " .. nouvelle_touche)
end

-- Restaurer les bindings sauvegardés au démarrage
local actions = { "bouger_gauche", "bouger_droite", "attaquer", "dash" }
for _, action in ipairs(actions) do
    local touche = Persist.get("bind_" .. action)
    if touche then
        InputBind.remap(action, touche)
    end
end
