Tutoriel : Ennemi avec IA

Créer un ennemi complet — patrouille, détection du joueur, poursuite, attaque, mort avec loot. Tous les modules clés en un seul exemple.

Ce que vous allez créer : un gobelin qui patrouille entre des points, détecte le joueur à 250 pixels, le poursuit, l'attaque à portée et lâche du butin en mourant.

1. Initialisation

lua  ·  ennemi_gobelin.lua — partie 1 : OnInit
-- === Variables du script (privées à chaque gobelin) ===
local sm               -- Machine à états
local joueur           -- Référence au joueur
local pv_max    = 40   -- Points de vie maximum
local degats    = 12   -- Dégâts par coup
local dist_vue  = 250  -- Distance de détection
local dist_atk  = 70   -- Distance d'attaque

local loot_table = {
    { id = "piece_or",    poids = 70, quantite = { min = 1, max = 3 } },
    { id = "potion_soin", poids = 20 },
    { id = "cle_rouille", poids = 10 },
}

function OnInit(entity)
    -- Trouver le joueur
    joueur = World.findByTag("player")

    -- Initialiser les stats
    Stats.init(entity.id, {
        Vie    = pv_max,
        VieMax = pv_max,
    })

    -- Réagir quand les PV tombent à 0
    EventBus.on("stat_zeroed_" .. tostring(entity.id), function(data)
        if data.stat == "Vie" then
            sm:setState("Mort")
        end
    end)

    -- Créer la machine à états (voir partie 2)
    sm = creerStateMachine(entity)
end

2. Machine à états

lua  ·  ennemi_gobelin.lua — partie 2 : états
function creerStateMachine(entity)
    return StateMachine.new({

        -- === PATROUILLE ===
        Patrouille = {
            onEnter = function()
                Anim.play(entity, "walk", true)
                -- Récupérer les points de patrol depuis l'éditeur
                local points = entity:getData("points_patrol")
                if points then Nav.patrol(entity, points) end
            end,
            onUpdate = function(dt)
                local dist = distanceAuJoueur(entity)
                if dist < dist_vue then
                    sm:setState("Poursuite")
                end
            end,
            onExit = function()
                Nav.stop(entity)
            end,
        },

        -- === POURSUITE ===
        Poursuite = {
            onEnter = function()
                Anim.play(entity, "run", true)
                Sound.play("data/sounds/gobelin_alerte.ogg")
            end,
            onUpdate = function(dt)
                local dist = distanceAuJoueur(entity)
                local pos_j = World.getPosition(joueur)

                if dist > dist_vue * 1.5 then
                    -- Le joueur est trop loin : retour en patrouille
                    sm:setState("Patrouille")
                elseif dist < dist_atk then
                    -- Le joueur est à portée : attaquer
                    sm:setState("Attaque")
                else
                    -- Continuer à poursuivre
                    Nav.moveTo(entity, pos_j.x, pos_j.y)
                end
            end,
        },

        -- === ATTAQUE ===
        Attaque = {
            onEnter = function()
                Nav.stop(entity)
                Anim.play(entity, "attack", false)

                -- Infliger des dégâts au bon frame
                Anim.onFrame(entity, 4, function()
                    if distanceAuJoueur(entity) < dist_atk * 1.2 then
                        Stats.mod(joueur.id, "Vie", -degats)
                        Camera.shake(0.15, 4)
                    end
                end)

                -- Revenir en poursuite après l'attaque
                Anim.onFinish(entity, function()
                    sm:setState("Poursuite")
                end)
            end,
        },

        -- === MORT ===
        Mort = {
            onEnter = function()
                Nav.stop(entity)
                Anim.play(entity, "death", false)
                Sound.play("data/sounds/gobelin_mort.ogg")

                -- Lâcher du butin
                local pos = World.getPosition(entity)
                Loot.drop(pos.x, pos.y, loot_table)

                -- Particules de mort
                Particles.emit(pos.x, pos.y, {
                    count = 15, speed = 80, spread = 360,
                    color = Color.new(100, 200, 80),
                })

                -- Désactiver l'entité après l'animation
                Anim.onFinish(entity, function()
                    EntityState.disable(entity)
                end)
            end,
        },

    }, "Patrouille")
end

3. Update et utilitaires

-- Fonction utilitaire : distance entre l'entité et le joueur
function distanceAuJoueur(entity)
    local pos_e = World.getPosition(entity)
    local pos_j = World.getPosition(joueur)
    return Math.distance(pos_e.x, pos_e.y, pos_j.x, pos_j.y)
end

function OnUpdate(entity, dt)
    -- Ne rien faire si le gobelin est mort
    if sm:is("Mort") then return end

    sm:update(dt)

    -- Barre de vie en debug
    Debug.watch("gobelin_vie", Stats.get(entity.id, "Vie", 0))
end

function OnDestroy(entity)
    Anim.clearCallbacks(entity)
end
Résultat : un gobelin avec 4 états, de l'animation synchronisée, des dégâts au bon frame, du loot à la mort, et un état persistant (ne réapparaît pas si tué).
