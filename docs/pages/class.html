Class

Créer des classes Lua réutilisables — orienté objet avec héritage prototype, constructeur et vérification de type.

Global automatique

API

Fonction / MéthodeDescription
Class.new()Crée une nouvelle classe sans parent.
Class.new(Parent)Crée une classe qui hérite de Parent.
Class.extendAlias de Class.new.
MaClasse.new(...)Crée une instance. Appelle :init(...) si défini. Notez le point, pas deux-points.
instance:is(Classe)Vrai si l'instance est de cette classe ou d'un parent.
Instantiation : utilisez MaClasse.new(...) (point) et non MaClasse:new(...) (deux-points). La méthode est un constructeur statique, pas une méthode d'instance.

Exemples

lua  ·  Classe simple
-- Définir une classe "Arme"
local Arme = Class.new()

function Arme:init(nom, degats)
    self.nom    = nom
    self.degats = degats
end

function Arme:attaquer(cible)
    Stats.mod(cible.id, "Vie", -self.degats)
    Log.info(self.nom .. " inflige " .. self.degats .. " dégâts !")
end

function Arme:getDescription()
    return self.nom .. " (" .. self.degats .. " dégâts)"
end

-- Créer des instances (point, pas deux-points)
local epee  = Arme.new("Épée longue",  25)
local hache = Arme.new("Hache de fer", 35)

epee:attaquer(ennemi)
Log.info(hache:getDescription())

lua  ·  Héritage
-- Sous-classe héritant de Arme
local ArmeEnchantee = Class.new(Arme)

function ArmeEnchantee:init(nom, degats, element)
    Arme.init(self, nom, degats)   -- appel du constructeur parent
    self.element = element
end

function ArmeEnchantee:attaquer(cible)
    Arme.attaquer(self, cible)     -- dégâts de base

    if self.element == "feu" then
        Effect.apply(cible.id, "brulure", { duration = 3.0,
            onTick = function(id, dt) Stats.mod(id, "Vie", -5 * dt) end,
        })
    elseif self.element == "glace" then
        Effect.apply(cible.id, "ralenti", { duration = 2.0,
            onApply  = function(id) Stats.mod(id, "Vitesse", -40) end,
            onExpire = function(id) Stats.mod(id, "Vitesse",  40) end,
        })
    end
end

local baton_feu  = ArmeEnchantee.new("Bâton de feu",  30, "feu")
local epee_glace = ArmeEnchantee.new("Épée de glace", 20, "glace")

baton_feu:attaquer(ennemi)

-- Vérification de type
Log.info(tostring(baton_feu:is(ArmeEnchantee)))  -- true
Log.info(tostring(baton_feu:is(Arme)))           -- true (héritage)

lua  ·  Classe autonome (sans héritage)
-- Compteur réutilisable
local Compteur = Class.new()

function Compteur:init(debut, pas)
    self.valeur = debut or 0
    self.pas    = pas   or 1
end

function Compteur:incrementer()
    self.valeur = self.valeur + self.pas
    return self.valeur
end

function Compteur:reset()
    self.valeur = 0
end

local score = Compteur.new(0, 10)
score:incrementer()  -- 10
score:incrementer()  -- 20
Log.info("Score : " .. tostring(score.valeur))
