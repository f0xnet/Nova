Random

Génération de nombres aléatoires — entiers, flottants, choix dans une liste, seed reproductible.

Global automatique

API

FonctionDescription
Random.int(min, max)Entier aléatoire entre min et max (inclus).
Random.float(min, max)Flottant aléatoire entre min et max.
Random.bool(probabilite)Vrai avec la probabilité donnée (0.0 à 1.0). Défaut : 0.5.
Random.pick(tableau)Retourne un élément aléatoire du tableau.
Random.shuffle(tableau)Mélange le tableau sur place.
Random.seed(valeur)Fixe la graine pour des résultats reproductibles.
Random.angle()Angle aléatoire en degrés (0 à 360).

Exemples

lua  ·  Usages courants
-- Dégâts aléatoires entre 15 et 25
local degats = Random.int(15, 25)
Stats.mod(ennemi.id, "Vie", -degats)

-- Position aléatoire dans une zone
local x = Random.float(100, 500)
local y = Random.float(100, 400)
World.setPosition(spawn, x, y)

-- Chance de 30% de lâcher un objet rare
if Random.bool(0.30) then
    Inventory.give(joueur, "gemme_rare")
    Notify.show("Objet rare trouvé !", "success")
end

-- Choisir une réplique au hasard
local repliques = {
    "Bonjour, aventurier !",
    "Beau temps pour se battre...",
    "Méfie-toi du boss au nord.",
}
local texte = Random.pick(repliques)
Conversation.start({{ speaker = "Garde", text = texte }})

lua  ·  Spawn aléatoire d'ennemis
local types_ennemis = { "gobelin", "squelette", "orc", "slime" }
local points_spawn  = World.findAllByTag("spawn_point")

-- Mélanger les points de spawn
Random.shuffle(points_spawn)

-- Spawner 3 ennemis aléatoires
for i = 1, 3 do
    local point  = points_spawn[i]
    local type_e = Random.pick(types_ennemis)
    local pos    = World.getPosition(point)

    World.spawn(type_e, pos.x, pos.y)
end

lua  ·  Seed reproductible (génération procédurale)
-- Générer le même donjon à chaque fois avec le même seed
local seed_donjon = 42
Random.seed(seed_donjon)

for i = 1, 10 do
    local x = Random.int(0, 20)
    local y = Random.int(0, 15)
    -- Toujours les mêmes positions avec seed = 42
    placerSalle(x, y)
end
