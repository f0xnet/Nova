Nav

Navigation automatique — déplacer une entité vers une cible en évitant les obstacles, suivre un chemin pré-calculé.

Global automatique

API

FonctionDescription
Nav.moveTo(entity, cible, vitesse, onFin)Déplace l'entité vers cible (entité ou Vec2f). vitesse et onFin sont optionnels.
Nav.findPath(entity, cible)Calcule et retourne le chemin vers la cible sous forme de liste de points [{x,y}...].
Nav.moveAlongPath(entity, chemin, vitesse, onFin)Déplace l'entité le long d'un chemin pré-calculé.
Nav.stop(entity)Arrête le déplacement de l'entité.
Nav.isMoving(entity)Vrai si l'entité est en déplacement.

Exemples

lua  ·  Déplacer un ennemi vers le joueur
function update(entity, dt)
    local joueur = World.findByTag("player")
    if not joueur then return end

    local dist = entity:getDistance(joueur)

    if dist < 300 then
        -- Poursuivre le joueur à vitesse 120
        Nav.moveTo(entity, joueur, 120)
        Anim.play(entity, "run", true)
    else
        Nav.stop(entity)
        Anim.play(entity, "idle", true)
    end
end

lua  ·  Déplacer vers une position fixe avec callback
-- Aller à un point, puis faire quelque chose à l'arrivée
local destination = World.findByTag("checkpoint_a"):getPosition()

Nav.moveTo(entity, destination, 100, function()
    print("Arrivé au checkpoint !")
    Anim.play(entity, "idle", true)
    Timer.after(2.0, function()
        -- Repartir vers le prochain point
        local suivant = World.findByTag("checkpoint_b"):getPosition()
        Nav.moveTo(entity, suivant, 100)
    end)
end)

lua  ·  Chemin manuel avec findPath
-- Calculer le chemin d'abord, puis le suivre
local joueur = World.findByTag("player")
local chemin = Nav.findPath(entity, joueur)

if #chemin > 0 then
    Nav.moveAlongPath(entity, chemin, 150, function()
        print("Chemin terminé !")
    end)

    -- Visualiser le chemin en debug
    for i = 1, #chemin - 1 do
        Debug.drawLine(
            chemin[i].x,   chemin[i].y,
            chemin[i+1].x, chemin[i+1].y,
            Color.new(0, 200, 255), 0
        )
    end
end

lua  ·  Patrouille entre plusieurs points
-- Patrouille manuelle entre des points de passage
local points = {
    World.findByTag("patrol_a"):getPosition(),
    World.findByTag("patrol_b"):getPosition(),
    World.findByTag("patrol_c"):getPosition(),
}
local index_actuel = 1

function allerAuProchainPoint()
    local destination = points[index_actuel]

    Nav.moveTo(entity, destination, 80, function()
        -- Passer au point suivant (boucle)
        index_actuel = (index_actuel % #points) + 1
        Timer.after(1.5, allerAuProchainPoint)
    end)
end

function init(entity)
    allerAuProchainPoint()
end
