Spatial

Recherches spatiales rapides — trouver les entités dans un rayon, dans un rectangle, ou les plus proches d'un point.

Global automatique

API

FonctionDescription
Spatial.findInRadius(x, y, rayon, tag)Retourne toutes les entités avec ce tag dans le rayon. Tag optionnel.
Spatial.findInRect(x, y, w, h, tag)Retourne toutes les entités dans le rectangle.
Spatial.findNearest(x, y, tag)Retourne l'entité la plus proche. Tag optionnel.
Spatial.findNNearest(x, y, n, tag)Retourne les n entités les plus proches.
Spatial.count(x, y, rayon, tag)Nombre d'entités dans le rayon (plus rapide que findInRadius).

Exemples

lua  ·  Zone d'explosion (dégâts en cercle)
function exploser(x, y)
    Particles.emit(x, y, { count = 40, speed = 250, spread = 360,
                            color = Color.new(255, 140, 0), gravity = 80 })
    Camera.shake(0.5, 15)
    Sound.play("data/sounds/explosion.ogg")

    -- Toucher tout ce qui est dans 120 pixels
    local victimes = Spatial.findInRadius(x, y, 120, "damageable")
    for _, victime in ipairs(victimes) do
        Stats.mod(victime.id, "Vie", -50)
    end
end

lua  ·  Cibler l'ennemi le plus proche
function attaquerLePlusProche(entity)
    local pos    = World.getPosition(entity)
    local cible  = Spatial.findNearest(pos.x, pos.y, "enemy")

    if cible then
        local dist = Math.distance(pos.x, pos.y,
                                    World.getPosition(cible).x,
                                    World.getPosition(cible).y)
        if dist < 150 then
            Stats.mod(cible.id, "Vie", -25)
        end
    end
end

lua  ·  Vérifier la densité ennemie
function OnUpdate(entity, dt)
    local pos = World.getPosition(entity)

    -- Changer de stratégie selon le nombre d'ennemis proches
    local nb_ennemis = Spatial.count(pos.x, pos.y, 300, "enemy")

    if nb_ennemis >= 5 then
        -- Trop d'ennemis : fuir
        sm:setState("Fuite")
    elseif nb_ennemis >= 2 then
        -- Quelques ennemis : appeler des renforts
        EventBus.emit("appel_renforts", { x = pos.x, y = pos.y })
    else
        -- Seul : attaquer normalement
        sm:setState("Attaque")
    end
end
