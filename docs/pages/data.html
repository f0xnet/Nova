Data

Lire des fichiers JSON de configuration — tableaux d'objets, stats de base, textes de dialogue. Idéal pour séparer les données du code.

Global automatique

API

FonctionDescription
Data.load(chemin)Lit un fichier JSON et retourne une table Lua. Mis en cache automatiquement.
Data.encode(table)Convertit une table Lua en texte JSON.
Data.decode(texte)Convertit un texte JSON en table Lua.

Exemples

json  ·  data/items.json
[
    { "id": "potion_soin",  "nom": "Potion de soin",  "soin": 30,  "prix": 15 },
    { "id": "epee_longue",  "nom": "Épée longue",     "degats": 25, "prix": 80 },
    { "id": "bouclier_fer", "nom": "Bouclier de fer", "defense": 10, "prix": 60 }
]

lua  ·  Charger et utiliser les données
-- Charger le fichier JSON
local items = Data.load("data/items.json")

-- Trouver un item par son id
local function trouverItem(id)
    for _, item in ipairs(items) do
        if item.id == id then
            return item
        end
    end
    return nil
end

-- Utiliser les données
local epee = trouverItem("epee_longue")
if epee then
    print("Dégâts de l'épée : " .. epee.degats)
    Stats.mod(ennemi.id, "Vie", -epee.degats)
end

-- Afficher le prix dans l'UI
local potion = trouverItem("potion_soin")
print("Prix : " .. potion.prix .. " pièces")

lua  ·  Sauvegarder une table dans Persist
-- Les tables Lua ne se sauvegardent pas directement
-- → Convertir en JSON d'abord

local progression = {
    niveau       = 5,
    experience   = 1240,
    competences  = { "feu", "glace" },
}

-- Sauvegarder
Persist.set("progression", Data.encode(progression))

-- Charger
local json         = Persist.get("progression", "{}")
local progression  = Data.decode(json)
print("Niveau : " .. progression.niveau)
