Anim

Contrôler les animations sprite — jouer, arrêter, changer de vitesse, déclencher des actions sur des frames précises.

Global automatique

API

FonctionDescription
Anim.play(entity, id, loop)Joue l'animation. loop = true par défaut. Repart du début si déjà en cours.
Anim.stop(entity)Arrête l'animation (reste sur le frame courant).
Anim.setSpeed(entity, speed)Vitesse de lecture. 1.0 = normale, 0.5 = ralenti, 2.0 = rapide.
Anim.current(entity)Nom de l'animation en cours.
Anim.onFrame(entity, frame, fn)Appelée une seule fois quand l'animation atteint le frame N.
Anim.onFinish(entity, fn)Appelée quand une animation non-boucle se termine.
Anim.clearCallbacks(entity)Supprime tous les callbacks de cette entité.

Exemple — Animation d'attaque

lua
function attaquer(entity)
    -- Jouer l'animation d'attaque (une seule fois, pas en boucle)
    Anim.play(entity, "attack", false)

    -- Au frame 3 : jouer le son de l'épée (synchronisé)
    Anim.onFrame(entity, 3, function()
        Sound.play("data/sounds/epee.ogg")
    end)

    -- Au frame 5 : zone de dégâts (le coup porte)
    Anim.onFrame(entity, 5, function()
        local pos = World.getPosition(entity)
        local victimes = Spatial.findInRadius(pos.x, pos.y, 60, "enemy")
        for _, e in ipairs(victimes) do
            Stats.mod(e.id, "Vie", -20)
        end
    end)

    -- Quand l'animation est terminée : retour à idle
    Anim.onFinish(entity, function()
        Anim.play(entity, "idle", true)
    end)
end


-- Changer l'animation selon le mouvement
function update(entity, dt)
    if Nav.isMoving(entity) then
        if Anim.current(entity) ~= "run" then
            Anim.play(entity, "run", true)
        end
    else
        if Anim.current(entity) ~= "idle" then
            Anim.play(entity, "idle", true)
        end
    end
end
