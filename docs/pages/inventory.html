Inventory

Inventaire par entité — ajouter, retirer, compter des objets, transférer entre entités.

Global automatique

API

FonctionDescription
Inventory.defineItem(item, def)Enregistre les propriétés d'un type d'objet (poids, maxStack, nom…). Optionnel.
Inventory.add(entityId, item, nb)Ajoute nb exemplaires (défaut : 1). Retourne la quantité ajoutée.
Inventory.remove(entityId, item, nb)Retire nb exemplaires. Retourne la quantité réellement retirée.
Inventory.count(entityId, item)Quantité possédée.
Inventory.has(entityId, item, nb)Vrai si l'entité possède au moins nb exemplaires (défaut : 1).
Inventory.getAll(entityId)Retourne { item = quantite, ... }.
Inventory.transfer(fromId, toId, item, nb)Transfert d'un inventaire à l'autre.
Inventory.clear(entityId)Vide complètement l'inventaire.
Inventory.clearItem(entityId, item)Retire toutes les copies d'un item.
Inventory.getWeight(entityId)Poids total (si les items ont un champ weight).

Événements émis

EventBus

item_added{ entityId, item, count, total }
item_removed{ entityId, item, count, total }

Exemples

lua  ·  Ramasser un objet au sol
-- Le joueur entre en contact avec un objet ramassable
function OnActivate(activatorId, actionID)
    local joueur = Registry:getEntity(activatorId)
    if joueur.tag ~= "player" then return end

    -- Ajouter l'objet à l'inventaire du joueur
    Inventory.add(joueur.id, "epee_longue", 1)
    Notify.show("Épée longue obtenue !", { color = "gold" })

    -- Désactiver l'objet au sol
    local entity = World.findByTag("item_epee")  -- ou via activatorId
    if entity then entity:disable() end
end

lua  ·  Coffre avec clé requise
function OnActivate(activatorId, actionID)
    local joueur = Registry:getEntity(activatorId)
    if joueur.tag ~= "player" then return end

    -- Vérifier si le joueur a la clé
    if not Inventory.has(joueur.id, "cle_donjon") then
        Notify.show("Il vous faut une clé.", { color = "orange" })
        return
    end

    -- Consommer la clé et ouvrir
    Inventory.remove(joueur.id, "cle_donjon", 1)
    Inventory.add(joueur.id, "epee_ancienne", 1)
    Inventory.add(joueur.id, "piece_or", 50)

    Notify.show("Coffre ouvert !", { color = "gold" })
    entity:disable()
end

lua  ·  Utiliser une potion
function utiliserPotion(joueur)
    if not Inventory.has(joueur.id, "potion_soin") then
        Notify.show("Pas de potions !", { color = "red" })
        return
    end

    -- Consommer la potion
    Inventory.remove(joueur.id, "potion_soin", 1)

    -- Effet de soin
    local soin = 50
    Stats.mod(joueur.id, "Vie", soin)
    Notify.floatingText(joueur, "+" .. soin .. " PV", { color = "lime" })
    Sound.play("data/sounds/potion.ogg")
end

lua  ·  Définir les propriétés d'un item
-- Définir au démarrage (script global)
Inventory.defineItem("epee_longue",  { weight = 3.0, maxStack = 1,  displayName = "Épée longue" })
Inventory.defineItem("potion_soin",  { weight = 0.5, maxStack = 10, displayName = "Potion de soin" })
Inventory.defineItem("piece_or",     { weight = 0.1, maxStack = 999, displayName = "Pièce d'or" })

-- Vérifier le poids total
local poids = Inventory.getWeight(joueur.id)
if poids > 20 then
    Notify.show("Inventaire trop lourd !", { color = "orange" })
end
