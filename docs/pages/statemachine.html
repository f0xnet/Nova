StateMachine

Machine à états — organiser le comportement d'un ennemi ou d'un objet en états distincts (Patrouille, Poursuite, Attaque, Mort…).

Global automatique

API

FonctionDescription
StateMachine.new(etat_initial)Crée une machine à états. Retourne un objet fsm.
fsm:addState(nom, onEnter, onUpdate, onExit)Ajoute un état. Les trois fonctions sont optionnelles.
fsm:update(entity, dt)À appeler dans OnUpdate — exécute l'état actuel.
fsm:transition(nom)Change d'état : appelle onExit de l'ancien, puis onEnter du nouveau.
fsm:getState()Nom de l'état actuel.

Exemple — IA d'ennemi simple

lua
local DISTANCE_DETECTION = 250
local DISTANCE_ATTAQUE   = 60

local fsm    -- La machine à états
local joueur

function OnInit(entity)
    joueur = World.findByTag("player")

    -- 1. Créer la machine avec l'état de départ
    fsm = StateMachine.new("Patrouille")

    -- 2. Ajouter chaque état
    fsm:addState("Patrouille",
        -- onEnter
        function()
            Anim.play(entity, "walk", true)
        end,
        -- onUpdate
        function(e, dt)
            local dist = e:getDistance(joueur)
            if dist < DISTANCE_DETECTION then
                fsm:transition("Poursuite")
            end
        end,
        -- onExit
        function()
            Nav.stop(entity)
        end
    )

    fsm:addState("Poursuite",
        function()
            Anim.play(entity, "run", true)
        end,
        function(e, dt)
            local dist = e:getDistance(joueur)
            if dist > DISTANCE_DETECTION * 1.5 then
                fsm:transition("Patrouille")
            elseif dist < DISTANCE_ATTAQUE then
                fsm:transition("Attaque")
            else
                local pos_j = joueur:getPosition()
                Nav.moveTo(entity, pos_j, 120)
            end
        end
    )

    fsm:addState("Attaque",
        function()
            Nav.stop(entity)
            Anim.play(entity, "attack", false)
            Anim.onFrame(entity, 4, function()
                Stats.mod(joueur.id, "Vie", -15)
            end)
            Anim.onFinish(entity, function()
                fsm:transition("Poursuite")
            end)
        end
    )
end

function OnUpdate(entity, dt)
    fsm:update(entity, dt)
end
