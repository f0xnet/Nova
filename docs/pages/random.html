Random

Génération de nombres aléatoires — entiers, flottants, choix dans une liste, dés, seed reproductible.

Global automatique

API

FonctionDescription
Random.int(min, max)Entier aléatoire entre min et max (inclus).
Random.float(min, max)Flottant aléatoire entre min et max.
Random.chance(probabilite)Vrai avec la probabilité donnée (0.0 à 1.0).
Random.pick(tableau)Retourne un élément aléatoire du tableau.
Random.shuffle(tableau)Mélange le tableau. Retourne le tableau mélangé.
Random.weighted(items)Tirage pondéré. Chaque item a un champ weight.
Random.dice(faces)Lance un dé à N faces. Ex : Random.dice(6) = 1 à 6.
Random.dice(nb, faces)Lance nb dés et retourne la somme. Ex : Random.dice(2, 6) = 2D6.
Random.gaussian(moyenne, ecart)Valeur aléatoire selon une courbe normale. Défauts : 0, 1.
Random.seed(valeur)Fixe la graine pour des résultats reproductibles.

Exemples

lua  ·  Usages courants
-- Dégâts aléatoires entre 15 et 25
local degats = Random.int(15, 25)
Stats.mod(ennemi.id, "Vie", -degats)

-- Position aléatoire dans une zone
local x = Random.float(100, 500)
local y = Random.float(100, 400)
entity:setPosition(x, y)

-- 30% de chance de lâcher un objet rare
if Random.chance(0.30) then
    Inventory.add(joueur.id, "gemme_rare")
    Notify.show("Objet rare trouvé !", "success")
end

-- Choisir une réplique au hasard
local repliques = {
    "Bonjour, aventurier !",
    "Beau temps pour se battre...",
    "Méfie-toi du boss au nord.",
}
local texte = Random.pick(repliques)

lua  ·  Dés et tirage pondéré
-- 2D6 pour les dégâts (style jeu de rôle)
local degats = Random.dice(2, 6)  -- entre 2 et 12
print("Dégâts : " .. degats)

-- Tirage pondéré pour le loot
local objets = {
    { value = "piece_or",    weight = 60 },
    { value = "potion_soin", weight = 25 },
    { value = "epee_rare",   weight = 10 },
    { value = "legendaire",  weight = 5  },
}
local gagne = Random.weighted(objets)
print("Obtenu : " .. gagne.value)

lua  ·  Seed reproductible
-- Générer le même donjon à chaque fois avec le même seed
Random.seed(42)

for i = 1, 10 do
    local x = Random.int(0, 20)
    local y = Random.int(0, 15)
    -- Toujours les mêmes positions avec seed = 42
    placerSalle(x, y)
end
